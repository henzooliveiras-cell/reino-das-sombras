using UnityEngine; using UnityEngine.SceneManagement; // CLASSE PRINCIPAL (O arquivo deve ter este nome: AMaldicaoDaVilaGame.cs) public class AMaldicaoDaVilaGame : MonoBehaviour { public static AMaldicaoDaVilaGame instancia; [Header("Configuração da Fase Atual")] public int totalInimigosNaFase; private int inimigosDerrotados = 0; public bool ehAFaseFinal = false; void Awake() { if (instancia == null) instancia = this; } public void InimigoDerrotado() { inimigosDerrotados++; Debug.Log($"Monstros mortos na floresta: {inimigosDerrotados}/{totalInimigosNaFase}"); if (inimigosDerrotados >= totalInimigosNaFase && !ehAFaseFinal) { AvancarFase(); } } void AvancarFase() { Debug.Log("Fase Limpa! Jason avança mais fundo na floresta destruída..."); SceneManager.LoadScene(SceneManager.GetActiveScene().buildIndex + 1); } public void VitoriaTotal() { Debug.Log("Fim da jornada. Jason derrotou o Dragão e a maldição da vila foi quebrada!"); } } // ========================================== // 1. SCRIPT DO JASON (PROTAGONISTA) // ========================================== [AddComponentMenu("A Maldição da Vila/Jason Controller")] public class JasonController : MonoBehaviour { [Header("Atributos do Jason")] public float vidaMaxima = 100f; private float vidaAtual; public float velocidadeMovimento = 5f; public float danoEspada = 25f; public float alcanceAtaque = 2f; [Header("Configuração de Combate")] public LayerMask camadaInimigos; private CharacterController controller; void Start() { vidaAtual = vidaMaxima; controller = GetComponent(); } void Update() { Mover(); // Clique esquerdo do mouse para atacar com a espada medieval if (Input.GetButtonDown("Fire1")) { AtacarComEspada(); } } void Mover() { float moverX = Input.GetAxis("Horizontal"); float moverZ = Input.GetAxis("Vertical"); Vector3 movimento = transform.right * moverX + transform.forward * moverZ; if (controller != null) { controller.Move(movimento * velocidadeMovimento * Time.deltaTime); } } void AtacarComEspada() { Debug.Log("Jason brandiu sua espada no vazio da floresta!"); // Detecta alvos no alcance do golpe frontal Collider[] alvosAtingidos = Physics.OverlapSphere(transform.position + transform.forward, alcanceAtaque, camadaInimigos); foreach (Collider inimigo in alvosAtingidos) { BichoAterrorizante bicho = inimigo.GetComponent(); if (bicho != null) bicho.ReceberDano(danoEspada); ChefaoDragao dragao = inimigo.GetComponent(); if (dragao != null) dragao.ReceberDano(danoEspada); } } public void ReceberDano(float dano) { vidaAtual -= dano; Debug.Log($"Jason ferido! Vida restante: {vidaAtual}"); if (vidaAtual <= 0) { Morrer(); } } void Morrer() { Debug.Log("Jason morreu. A floresta escura consumiu sua alma..."); SceneManager.LoadScene(SceneManager.GetActiveScene().buildIndex); // Reinicia a fase } private void OnDrawGizmosSelected() { Gizmos.color = Color.red; Gizmos.DrawWireSphere(transform.position + transform.forward, alcanceAtaque); } } // ========================================== // 2. SCRIPT DOS BICHOS ATERRORIZANTES // ========================================== [AddComponentMenu("A Maldição da Vila/Bicho Aterrorizante")] public class BichoAterrorizante : MonoBehaviour { [Header("Status do Monstro")] public float vidaMaxima = 50f; private float vidaAtual; public float danoAtaque = 15f; public float velocidade = 3.5f; public float distanciaAtaque = 1.5f; private Transform alvoJason; void Start() { vidaAtual = vidaMaxima; GameObject player = GameObject.FindGameObjectWithTag("Player"); if (player != null) alvoJason = player.transform; } void Update() { if (alvoJason == null) return; float distancia = Vector3.Distance(transform.position, alvoJason.position); // Persegue Jason pelo mundo solitário if (distancia > distanciaAtaque) { transform.LookAt(alvoJason); transform.position = Vector3.MoveTowards(transform.position, alvoJason.position, velocidade * Time.deltaTime); } else { AtacarJason(); } } void AtacarJason() { JasonController jason = alvoJason.GetComponent(); if (jason != null) { jason.ReceberDano(danoAtaque * Time.deltaTime); } } public void ReceberDano(float dano) { vidaAtual -= dano; Debug.Log($"Bicho Aterrorizante urrou de dor! Vida: {vidaAtual}"); if (vidaAtual <= 0) { if (AMaldicaoDaVilaGame.instancia != null) AMaldicaoDaVilaGame.instancia.InimigoDerrotado(); Destroy(gameObject); } } } // ========================================== // 3. SCRIPT DO CHEFÃO DRAGÃO MEDIEVAL // ========================================== [AddComponentMenu("A Maldição da Vila/Chefão Dragão")] public class ChefaoDragao : MonoBehaviour { [Header("Status do Dragão")] public float vidaMaxima = 500f; private float vidaAtual; public float danoFogo = 40f; public float alcanceSoproFogo = 8f; private Transform alvoJason; void Start() { vidaAtual = vidaMaxima; GameObject player = GameObject.FindGameObjectWithTag("Player"); if (player != null) alvoJason = player.transform; } void Update() { if (alvoJason == null) return; // O dragão sempre encara Jason de forma intimidadora transform.LookAt(alvoJason); float distancia = Vector3.Distance(transform.position, alvoJason.position); if (distancia <= alcanceSoproFogo) { SoprarFogo(); } } void SoprarFogo() { Debug.Log("O Dragão comanda a escuridão e sopra chamas medievais em Jason!"); JasonController jason = alvoJason.GetComponent(); if (jason != null) { jason.ReceberDano(danoFogo * Time.deltaTime); } } public void ReceberDano(float dano) { vidaAtual -= dano; Debug.Log($"O Dragão soberano sofreu dano! Vida: {vidaAtual}"); if (vidaAtual <= 0) { Morrer(); } } void Morrer() { if (AMaldicaoDaVilaGame.instancia != null) AMaldicaoDaVilaGame.instancia.VitoriaTotal(); Destroy(gameObject); } }
